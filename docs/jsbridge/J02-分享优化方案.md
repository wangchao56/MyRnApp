## 现有代码分析

| 优点 | 不足 |
|------|------|
| Promise 化异步调用 | `isInApp()` 只判断 RN WebView，未覆盖微信环境 |
| 超时机制 | `getMockResult` 硬编码，无法扩展 |
| App 事件订阅/取消 | 环境判断单一，分享未做环境分流 |
| msgId 追踪回调 | 缺少微信 H5/小程序的降级处理 |

// env.ts
export class EnvDetector {
  static isRNWebView(): boolean {
    return typeof window !== 'undefined' && !!window.ReactNativeWebView;
  }

  static isWechat(): boolean {
    return typeof navigator !== 'undefined' && /MicroMessenger/i.test(navigator.userAgent);
  }

  static async isMiniProgram(): Promise<boolean> {
    if (!EnvDetector.isWechat()) return false;
    
    return new Promise((resolve) => {
      if (window.wx?.miniProgram) {
        wx.miniProgram.getEnv((res: any) => resolve(!!res.miniprogram));
      } else {
        // 延迟检测，等待 wx.config 完成
        setTimeout(() => {
          resolve(window.wx?.miniProgram?.getEnv ? false : false);
        }, 100);
      }
    });
  }

  static async getPlatform(): Promise<<PlatformType> {
    if (EnvDetector.isRNWebView()) return 'rn';
    if (await EnvDetector.isMiniProgram()) return 'miniprogram';
    if (EnvDetector.isWechat()) return 'wechat-h5';
    return 'browser';
  }
}

import { ShareOptions, PlatformType } from './types';

export interface ShareAdapter {
  share(options: ShareOptions): Promise<any>;
  isAvailable(): boolean | Promise<boolean>;
}

// RN 分享适配器
export class RNShareAdapter implements ShareAdapter {
  private sendMessage: (msg: string) => void;

  constructor(sendMessage: (msg: string) => void) {
    this.sendMessage = sendMessage;
  }

  isAvailable(): boolean {
    return typeof window !== 'undefined' && !!window.ReactNativeWebView;
  }

  async share(options: ShareOptions): Promise<any> {
    return new Promise((resolve, reject) => {
      const msgId = `${Date.now()}_${Math.random().toString(36).slice(2)}`;
      
      // 注册一次性回调
      const handler = (event: MessageEvent) => {
        try {
          const data = JSON.parse(event.data);
          if (data.msgId === msgId) {
            window.removeEventListener('message', handler);
            data.code === 200 ? resolve(data.data) : reject(new Error(data.error));
          }
        } catch {}
      };
      
      window.addEventListener('message', handler);
      
      this.sendMessage(JSON.stringify({
        msgId,
        action: 'share',
        data: options
      }));
    });
  }
}

// 微信 JSSDK 适配器
export class WechatJSSDKAdapter implements ShareAdapter {
  private wxReady: Promise<void>;

  constructor() {
    this.wxReady = new Promise((resolve) => {
      if (window.wx?.ready) {
        window.wx.ready(resolve);
      } else {
        resolve(); // 未配置 JSSDK 时直接 resolve
      }
    });
  }

  isAvailable(): boolean {
    return typeof window !== 'undefined' && !!window.wx;
  }

  async share(options: ShareOptions): Promise<any> {
    await this.wxReady;
    
    const shareData = {
      title: options.title,
      desc: options.desc || '',
      link: options.link,
      imgUrl: options.imgUrl || '',
      success: () => {},
      cancel: () => {}
    };

    return new Promise((resolve, reject) => {
      const success = () => resolve({ success: true });
      const cancel = () => resolve({ success: false, cancelled: true });
      
      window.wx.updateAppMessageShareData({
        ...shareData,
        success,
        cancel
      });
      
      window.wx.updateTimelineShareData({
        title: options.title,
        link: options.link,
        imgUrl: options.imgUrl,
        success,
        cancel
      });
      
      // 微信 H5 无法主动唤起分享面板，返回提示信息
      resolve({ 
        success: true, 
        method: 'wechat-menu',
        message: '请点击右上角分享'
      });
    });
  }
}

// 小程序 WebView 适配器
export class MiniProgramAdapter implements ShareAdapter {
  isAvailable(): boolean {
    return typeof window !== 'undefined' && !!window.wx?.miniProgram;
  }

  async share(options: ShareOptions): Promise<any> {
    return new Promise((resolve) => {
      window.wx.miniProgram.postMessage({
        data: {
          action: 'share',
          ...options
        }
      });
      
      // 通知小程序显示分享菜单
      window.wx.miniProgram.showShareMenu({
        withShareTicket: true,
        menus: ['shareAppMessage', 'shareTimeline']
      });
      
      resolve({ 
        success: true, 
        method: 'miniprogram-menu',
        message: '请点击右上角分享'
      });
    });
  }
}

// 浏览器原生分享适配器
export class BrowserShareAdapter implements ShareAdapter {
  isAvailable(): boolean {
    return typeof navigator !== 'undefined' && !!navigator.share;
  }

  async share(options: ShareOptions): Promise<any> {
    return navigator.share({
      title: options.title,
      text: options.desc,
      url: options.link
    });
  }
}

// 降级：复制链接适配器
export class ClipboardAdapter implements ShareAdapter {
  isAvailable(): boolean {
    return typeof navigator !== 'undefined' && !!navigator.clipboard;

  async share(options: ShareOptions): Promise<any> {
    await navigator.clipboard.writeText(`${options.title}\n${options.link}`);
    return { success: true, method: 'clipboard', message: '链接已复制' };
  }

```typescript
// jsbridge.ts（优化后的主类）
import { 
  BridgeRequest, 
  BridgeResponse, 
  DEFAULT_TIMEOUT, 
  ShareOptions,
  PlatformType 
} from './types';
import { EnvDetector } from './env';
import {
  ShareAdapter,
  RNShareAdapter,
  WechatJSSDKAdapter,
  MiniProgramAdapter,
  BrowserShareAdapter,
  ClipboardAdapter
} from './share-adapters';

type PendingCallback = {
  resolve: (value: any) => void;
  reject: (error: Error) => void;
  timeoutId?: ReturnType<<typeof setTimeout>;
};

interface JSBridgeOptions {
  timeout?: number;
  enableMock?: boolean;      // 是否启用 mock
  mockHandler?: MockHandler;  // 自定义 mock 逻辑
}

type MockHandler = (action: string, data?: any) => any;

class JSBridge {
  private callbacks: Map<string, PendingCallback> = new Map();
  private idCounter = 0;
  private timeout: number;
  private enableMock: boolean;
  private mockHandler?: MockHandler;
  private isInitialized = false;
  private platform: PlatformType | null = null;
  private shareAdapters: Map<<PlatformType, ShareAdapter[]> = new Map();

  constructor(options: JSBridgeOptions = {}) {
    this.timeout = options.timeout ?? DEFAULT_TIMEOUT;
    this.enableMock = options.enableMock ?? false;
    this.mockHandler = options.mockHandler;
    this.init();
  }

  private async init(): Promise<void> {
    this.setupGlobalReceiver();
    this.platform = await EnvDetector.getPlatform();
    this.setupShareAdapters();
    this.isInitialized = true;
  }

  private generateMsgId(): string {
    return `${Date.now()}_${++this.idCounter}_${Math.random().toString(36).slice(2, 6)}`;
  }

  private setupGlobalReceiver(): void {
    if (typeof window === 'undefined') return;
    
    (window as any).__RECEIVE_MESSAGE_FROM_APP__ = this.receiveMessage.bind(this);
    (window as any).__ON_APP_EVENT__ = this.handleAppEvent.bind(this);
  }

  // ========== 环境检测 ==========
  
  getPlatform(): PlatformType | null {
    return this.platform;
  }

  isInApp(): boolean {
    return this.platform === 'rn';
  }

  isInMiniProgram(): boolean {
    return this.platform === 'miniprogram';
  }

  isInWechat(): boolean {
    return this.platform === 'wechat-h5' || this.platform === 'miniprogram';
  }

  // ========== 核心调用 ==========

  invoke<T = any, R = any>(action: string, data?: T): Promise<R> {
    return new Promise((resolve, reject) => {
      // Mock 模式：不在 App 内时返回 mock 数据
      if (!this.isInApp()) {
        if (this.enableMock) {
          const mockResult = this.getMockResult(action, data);
          if (mockResult !== undefined) {
            resolve(mockResult as R);
            return;
          }
        }
        reject(new Error(`Action '${action}' 需要在 App 环境内执行，当前环境: ${this.platform}`));
        return;
      }

      const msgId = this.generateMsgId();
      
      const timeoutId = setTimeout(() => {
        this.callbacks.delete(msgId);
        reject(new Error(`Action '${action}' 超时 (${this.timeout}ms)`));
      }, this.timeout);

      this.callbacks.set(msgId, { resolve, reject, timeoutId });

      const request: BridgeRequest<T> = { msgId, action, data };
      window.ReactNativeWebView?.postMessage(JSON.stringify(request));
    });
  }

  private receiveMessage(responseString: string): void {
    try {
      const response: BridgeResponse = JSON.parse(responseString);
      const { msgId, code, data, error } = response;

      const callback = this.callbacks.get(msgId);
      if (!callback) return;

      if (callback.timeoutId) clearTimeout(callback.timeoutId);

      if (code === 200 || code === 0) {
        callback.resolve(data);
      } else {
        callback.reject(new Error(error || `Error code: ${code}`));
      }

      this.callbacks.delete(msgId);
    } catch (e) {
      console.error('[JSBridge] 解析 App 返回数据失败', e);
    }
  }

  // ========== 分享功能（多环境适配） ==========

  private setupShareAdapters(): void {
    if (!this.platform) return;

    const adapters: ShareAdapter[] = [];
    
    switch (this.platform) {
      case 'rn':
        adapters.push(new RNShareAdapter(
          (msg) => window.ReactNativeWebView?.postMessage(msg)
        ));
        break;
        
      case 'miniprogram':
        adapters.push(new MiniProgramAdapter());
        break;
        
      case 'wechat-h5':
        adapters.push(new WechatJSSDKAdapter());
        break;
        
      case 'browser':
        adapters.push(new BrowserShareAdapter());
        adapters.push(new ClipboardAdapter()); // 降级
        break;
    }

    this.shareAdapters.set(this.platform, adapters);
  }

  /**
   * 统一分享入口
   */
  async share(options: ShareOptions): Promise<any> {
    // 如果在 App 内，优先使用原生分享
    if (this.isInApp()) {
      return this.invoke('share', options);
    }

    // 其他环境使用适配器链
    const adapters = this.shareAdapters.get(this.platform!) || [];
    
    for (const adapter of adapters) {
      const available = await adapter.isAvailable();
      if (available) {
        try {
          return await adapter.share(options);
        } catch (error) {
          console.warn(`[JSBridge] 分享适配器失败，尝试下一个`, error);
          continue;
        }
      }
    }

    // 最终降级：复制到剪贴板
    if (navigator.clipboard) {
      await navigator.clipboard.writeText(`${options.title}\n${options.link}`);
      return { success: true, method: 'clipboard-fallback', message: '链接已复制到剪贴板' };
    }

    throw new Error('当前环境不支持分享功能');
  }

  // ========== App 事件订阅 ==========

  onAppEvent(eventType: string, handler: (data: any) => void): () => void {
    if (typeof window === 'undefined') return () => {};

    const handlers = (window as any).__APP_EVENT_HANDLERS__ ||= {};
    handlers[eventType] = handler;

    // 返回取消订阅函数
    return () => this.offAppEvent(eventType);
  }

  offAppEvent(eventType: string): void {
    if (typeof window !== 'undefined' && (window as any).__APP_EVENT_HANDLERS__) {
      delete (window as any).__APP_EVENT_HANDLERS__[eventType];
    }
  }

  private handleAppEvent(eventType: string, data: any): void {
    const handlers = (window as any).__APP_EVENT_HANDLERS__;
    handlers?.[eventType]?.(data);
  }

  // ========== Mock 系统 ==========

  private getMockResult(action: string, data?: any): any {
    // 优先使用自定义 mockHandler
    if (this.mockHandler) {
      const custom = this.mockHandler(action, data);
      if (custom !== undefined) return custom;
    }

    // 默认 mock 数据
    const mocks: Record<string, any> = {
      'getUserInfo': { userId: '888', name: 'Demo User', avatar: '' },
      'getToken': 'mock_token_' + Date.now(),
      'getLocation': { latitude: 31.2304, longitude: 121.4737, city: '上海' },
      'scanQRCode': { result: 'https://example.com', format: 'QR_CODE' },
      'share': { success: true, mock: true },
      'getDeviceInfo': { 
        platform: 'iOS', 
        version: '16.0', 
        model: 'iPhone14,2',
        appVersion: '1.0.0'
      }
    };

    return mocks[action];
  }

  // ========== 工具方法 ==========

  get pendingCount(): number {
    return this.callbacks.size;
  }

  destroy(): void {
    // 清理所有 pending 回调
    this.callbacks.forEach(({ timeoutId, reject }) => {
      if (timeoutId) clearTimeout(timeoutId);
      reject(new Error('JSBridge 已销毁'));
    });
    this.callbacks.clear();
  }
}

// 单例导出
export const jsbridge = new JSBridge({
  enableMock: process.env.NODE_ENV === 'development'
});

export const invoke = <T = any, R = any>(action: string, data?: T): Promise<R> => {
  return jsbridge.invoke<T, R>(action, data);
};

export const share = (options: ShareOptions): Promise<any> => {
  return jsbridge.share(options);
};

export const isInApp = (): boolean => jsbridge.isInApp();
export const getPlatform = (): PlatformType | null => jsbridge.getPlatform();

export default jsbridge;
```

// ReactNativeWebView.tsx
import WebView from 'react-native-webview';
import * as WeChat from 'react-native-wechat-lib';

const ShareWebView = ({ source }: { source: { uri: string } }) => {
  const webViewRef = useRef<WebView>(null);

  const handleMessage = async (event: any) => {
    const request = JSON.parse(event.nativeEvent.data);
    const { msgId, action, data } = request;

    let response: any;

    try {
      switch (action) {
        case 'share':
          response = await handleShare(data);
          break;
        case 'getUserInfo':
          response = await getUserInfo();
          break;
        // ... 其他 action
        default:
          throw new Error(`未知 action: ${action}`);
      }
    } catch (error: any) {
      response = { code: 500, error: error.message };
    }

    // 回传结果
    webViewRef.current?.postMessage(JSON.stringify({
      msgId,
      code: response.code || 200,
      data: response.data || response,
      error: response.error
    }));
  };

  const handleShare = async (options: any) => {
    const { title, desc, link, imgUrl, type = 'link', scene = 0 } = options;
    
    // 调用微信分享
    const result = await WeChat.shareWebpage({
      title,
      description: desc,
      thumbImageUrl: imgUrl,
      webpageUrl: link,
      scene
    });
    
    return { success: true, platformResult: result };
  };

  return (
    <WebView
      ref={webViewRef}
      source={source}
      onMessage={handleMessage}
      javaScriptEnabled
      allowUniversalAccessFromFileURLs
    />
  );

## 关键优化点总结

| 优化项 | 实现 |
|--------|------|
| **多环境检测** | `EnvDetector` 类，支持 RN / 微信 H5 / 小程序 / 浏览器 |
| **分享适配器模式** | `ShareAdapter` 接口 + 各平台实现类 |
| **自动降级链** | 浏览器环境：Web Share API → Clipboard |
| **可扩展 Mock** | 构造函数注入 `mockHandler`，支持自定义 |
| **取消订阅** | `onAppEvent` 返回取消函数，避免内存泄漏 |
| **销毁清理** | `destroy()` 方法，终止所有 pending Promise |
| **msgId 增强** | 加入随机数，避免并发冲突 |
