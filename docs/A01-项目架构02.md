# Monorepo 多平台统一架构

## 一、架构概览

### 1.1 设计目标

- **一套代码三平台**：Android、iOS、Web 统一开发
- **独立开发**：App 原生项目和 Web 项目可以独立迭代
- **代码共享**：业务逻辑、组件、API 层在共享包中统一维护
- **WebView 嵌入**：支持 App 内嵌 Web 页面的开发模式
- **状态管理**：使用 MST (MobX State Tree) 进行统一状态管理

### 4.2 平台适配工具 `packages/shared/src/utils/platform.ts`

```typescript
import { Platform, Dimensions } from 'react-native';

export const isWeb = Platform.OS === 'web';
export const isNative = Platform.OS !== 'web';
export const isIOS = Platform.OS === 'ios';
export const isAndroid = Platform.OS === 'android';

export const getScreenSize = () => {
  const { width, height } = Dimensions.get('window');
  return { width, height };
};

export function platformSelect<T>(options: {
  ios?: T;
  android?: T;
  web?: T;
  default?: T;
}): T | undefined {
  if (isIOS && options.ios !== undefined) return options.ios;
  if (isAndroid && options.android !== undefined) return options.android;
  if (isWeb && options.web !== undefined) return options.web;
  return options.default;
}

export const getSafeAreaInsets = () => {
  if (isWeb) {
    const bodyStyle = window.getComputedStyle(document.body);
    return {
      top: parseInt(bodyStyle.paddingTop) || 0,
      bottom: parseInt(bodyStyle.paddingBottom) || 0,
      left: 0,
      right: 0,
    };
  }
  return null;
};
```

```typescript
const webStorage: StorageAdapter = {
  getItem: async (key) => localStorage.getItem(key),
  setItem: async (key, value) => localStorage.setItem(key, value),
  removeItem: async (key) => localStorage.removeItem(key),
  clear: async () => localStorage.clear(),
};

let nativeStorage: StorageAdapter | null = null;

const getNativeStorage = async (): Promise<StorageAdapter> => {
  if (!nativeStorage) {
    const AsyncStorage = require('@react-native-async-storage/async-storage').default;
    nativeStorage = {
      getItem: (key) => AsyncStorage.getItem(key),
      setItem: (key, value) => AsyncStorage.setItem(key, value),
      removeItem: (key) => AsyncStorage.removeItem(key),
      clear: () => AsyncStorage.clear(),
    };
  }
  return nativeStorage;
};

export const Storage = {
  async getItem(key: string): Promise<string | null> {
    if (Platform.OS === 'web') {
      return webStorage.getItem(key);
    }
    const storage = await getNativeStorage();
    return storage.getItem(key);
  },

  async setItem(key: string, value: StorageValue): Promise<void> {
    const stringValue = typeof value === 'string' ? value : JSON.stringify(value);
    if (Platform.OS === 'web') {
      return webStorage.setItem(key, stringValue);
    }
    const storage = await getNativeStorage();
    return storage.setItem(key, stringValue);
  },

  async removeItem(key: string): Promise<void> {
    if (Platform.OS === 'web') {
      return webStorage.removeItem(key);
    }
    const storage = await getNativeStorage();
    return storage.removeItem(key);
  },

  async clear(): Promise<void> {
    if (Platform.OS === 'web') {
      return webStorage.clear();
    }
    const storage = await getNativeStorage();
    return storage.clear();
  },

  async getJSON<T>(key: string): Promise<T | null> {
    const value = await this.getItem(key);
    if (!value) return null;
    try {
      return JSON.parse(value) as T;
    } catch {
      return null;
    }
  },
};
```

```typescript
import { api } from '../services/api';

export const UserModel = types.model('User', {
  id: types.identifier,
  name: types.string,
  email: types.string,
  avatar: types.maybe(types.string),
  token: types.maybe(types.string),
});

export type User = Instance<typeof UserModel>;

export const UserStore = types
  .model('UserStore', {
    user: types.maybe(UserModel),
    isLoggedIn: types.boolean,
    isLoading: types.boolean,
    error: types.maybe(types.string),
  })
  .actions((self) => ({
    login: flow(function* (email: string, password: string) {
      self.isLoading = true;
      self.error = undefined;
      try {
        const response = yield api.post('/auth/login', { email, password });
        const userData = response.data;
        self.user = UserModel.create(userData);
        self.isLoggedIn = true;
        yield Storage.setItem('user', userData);
        yield Storage.setItem('token', userData.token);
      } catch (err) {
        self.error = err instanceof Error ? err.message : 'Login failed';
        throw err;
      } finally {
        self.isLoading = false;
      }
    }),

    logout: flow(function* () {
      try {
        yield api.post('/auth/logout');
      } catch {
      } finally {
        self.user = undefined;
        self.isLoggedIn = false;
        yield Storage.removeItem('user');
        yield Storage.removeItem('token');
      }
    }),

    loadFromStorage: flow(function* () {
      const userData = yield Storage.getJSON<User>('user');
      if (userData) {
        self.user = UserModel.create(userData);
        self.isLoggedIn = true;
      }
    }),

    updateUser: flow(function* (userData: Partial<User>) {
      if (!self.user) return;
      Object.assign(self.user, userData);
      yield Storage.setItem('user', self.user);
    }),
  }))
  .views((self) => ({
    get displayName() {
      return self.user?.name || 'Guest';
    },
  }));
```

### 5.3 WebView 容器 `packages/app/src/components/WebViewContainer.tsx`

```typescript
import React, { useRef, useCallback } from 'react';
import { View, StyleSheet, Platform } from 'react-native';
import { WebView, WebViewNavigation } from 'react-native-webview';
import { useUserStore } from '@myapp/shared';

interface WebViewContainerProps {
  uri: string;
  onMessage?: (data: any) => void;
  onNavigationStateChange?: (navState: WebViewNavigation) => void;
}

export const WebViewContainer: React.FC<WebViewContainerProps> = ({
  uri,
  onMessage,
  onNavigationStateChange,
}) => {
  const webViewRef = useRef<WebView>(null);
  const userStore = useUserStore();

  const handleMessage = useCallback(
    (event: any) => {
      try {
        const data = JSON.parse(event.nativeEvent.data);

        switch (data.type) {
          case 'getToken':
            webViewRef.current?.postMessage(
              JSON.stringify({
                type: 'tokenResponse',
                token: userStore.user?.token || '',
              })
            );
            break;

          case 'getUserInfo':
            webViewRef.current?.postMessage(
              JSON.stringify({
                type: 'userInfoResponse',
                user: userStore.user,
              })
            );
            break;

          case 'goBack':
            webViewRef.current?.goBack();
            break;

          default:
            onMessage?.(data);
        }
      } catch (error) {
        console.error('WebView message parse error:', error);
      }
    },
    [userStore, onMessage]
  );

  const injectedJavaScript = `
    (function() {
      window.JSBridge = {
        postMessage: function(data) {
          window.ReactNativeWebView.postMessage(JSON.stringify(data));
        }
      };

      window.addEventListener('message', function(event) {
        try {
          var data = typeof event.data === 'string' ? JSON.parse(event.data) : event.data;
          if (window.JSBridge._callbacks && window.JSBridge._callbacks[data.type]) {
            window.JSBridge._callbacks[data.type](data);
          }
        } catch(e) {}
      });

      window.JSBridge._callbacks = {};
      window.JSBridge.onMessage = function(type, callback) {
        this._callbacks[type] = callback;
      };

      window.JSBridge.getToken = function() {
        return new Promise(function(resolve) {
          window.JSBridge.postMessage({ type: 'getToken' });
          window.JSBridge.onMessage('tokenResponse', function(data) {
            resolve(data.token);
          });
        });
      };

      window.JSBridge.getUserInfo = function() {
        return new Promise(function(resolve) {
          window.JSBridge.postMessage({ type: 'getUserInfo' });
          window.JSBridge.onMessage('userInfoResponse', function(data) {
            resolve(data.user);
          });
        });
      };
    })();
    true;
  `;

  return (
    <View style={styles.container}>
      <WebView
        ref={webViewRef}
        source={{ uri }}
        style={styles.webView}
        injectedJavaScript={injectedJavaScript}
        onMessage={handleMessage}
        onNavigationStateChange={onNavigationStateChange}
        javaScriptEnabled={true}
        domStorageEnabled={true}
        startInLoadingState={true}
        scalesPageToFit={true}
        allowsInlineMediaPlayback={true}
        mediaPlaybackRequiresUserAction={false}
        {...(Platform.OS === 'android' && {
          androidLayerType: 'software',
          mixedContentMode: 'compatibility',
        })}
      />
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
  },
  webView: {
    flex: 1,
  },
});
```

### 6.3 JSBridge 实现 `packages/web/src/jsbridge/index.ts`

```typescript
export const isInWebView = (): boolean => {
  return typeof window !== 'undefined' && !!window.ReactNativeWebView;
};

interface JSBridgeInterface {
  isInWebView: () => boolean;
  getToken: () => Promise<string>;
  getUserInfo: () => Promise<any>;
  postMessage: (data: any) => void;
  onMessage: (type: string, callback: (data: any) => void) => void;
}

declare global {
  interface Window {
    ReactNativeWebView?: {
      postMessage: (message: string) => void;
    };
    JSBridge?: {
      postMessage: (data: any) => void;
      onMessage: (type: string, callback: (data: any) => void) => void;
      getToken: () => Promise<string>;
      getUserInfo: () => Promise<any>;
    };
  }
}

export const JSBridge: JSBridgeInterface = {
  isInWebView,

  postMessage: (data: any) => {
    if (window.ReactNativeWebView) {
      window.ReactNativeWebView.postMessage(JSON.stringify(data));
    }
  },

  onMessage: (type: string, callback: (data: any) => void) => {
    if (window.JSBridge) {
      window.JSBridge.onMessage(type, callback);
    }
  },

  getToken: async (): Promise<string> => {
    if (isInWebView() && window.JSBridge) {
      return window.JSBridge.getToken();
    }
    return localStorage.getItem('token') || '';
  },

  getUserInfo: async () => {
    if (isInWebView() && window.JSBridge) {
      return window.JSBridge.getUserInfo();
    }
    const userStr = localStorage.getItem('user');
    return userStr ? JSON.parse(userStr) : null;
  },
};

export const initJSBridge = () => {
  if (!isInWebView()) {
    window.JSBridge = {
      postMessage: (data) => console.log('JSBridge mock:', data),
      onMessage: () => {},
      getToken: async () => localStorage.getItem('token') || '',
      getUserInfo: async () => {
        const userStr = localStorage.getItem('user');
        return userStr ? JSON.parse(userStr) : null;
      },
    };
  }
};

export default JSBridge;
```

## 八、架构优势总结

| 特性 | 说明 |
|------|------|
| **MST + MobX** | 使用 MobX State Tree 进行状态管理，支持异步操作、持久化、类型安全 |
| **Turborepo** | 并行构建加速，增量缓存，跨包依赖管理 |
| **独立开发** | App 和 Web 是独立的包，可以独立发版、独立测试 |
| **代码共享** | 共享层 `@myapp/shared` 被 App 和 Web 共同依赖 |
| **实时同步** | 修改共享层代码，App 和 Web 同时生效 |
| **隔离差异** | 平台特有代码在各自包内实现，不互相污染 |
| **独立部署** | Web 可以部署到 CDN，App 内 WebView 加载远程地址 |
| **CI/CD** | GitHub Actions 自动化测试和构建 |
