```typescript
import {types, Instance, SnapshotOut} from 'mobx-state-tree';

const UserModel = types
  .model('User')
  .props({
    id: types.maybe(types.number),
    name: types.maybe(types.string),
    token: types.maybe(types.string),
  })
  .actions(self => ({
    setUser(data: {id: number; name: string; token: string}) {
      self.id = data.id;
      self.name = data.name;
      self.token = data.token;
    },
    clearUser() {
      self.id = undefined;
      self.name = undefined;
      self.token = undefined;
    },
  }));

export const RootStoreModel = types
  .model('RootStore')
  .props({
    user: types.optional(UserModel, {}),
  })
  .actions(self => ({
    afterCreate() {
      // 初始化逻辑
    },
  }));

export interface RootStore extends Instance<typeof RootStoreModel> {}
export interface RootStoreSnapshot extends SnapshotOut<typeof RootStoreModel> {}
```

创建 `app/models/root-store-context.tsx`：

```typescript
import React, {createContext, useContext} from 'react';
import {RootStore} from './root-store';

const RootStoreContext = createContext<RootStore | undefined>(undefined);

export const RootStoreProvider: React.FC<{
  value: RootStore;
  children: React.ReactNode;
}> = ({value, children}) => (
  <RootStoreContext.Provider value={value}>
    {children}
  </RootStoreContext.Provider>
);

export const useStores = () => {
  const context = useContext(RootStoreContext);
  if (!context) {
    throw new Error('useStores must be used within RootStoreProvider');
  }
  return context;
};
```

#### 步骤 6：创建 API 服务

创建 `app/services/api/api.ts`：

```typescript
import {create, ApisauceInstance, ApiResponse} from 'apisauce';

const DEFAULT_CONFIG = {
  url: 'https://api.example.com',
  timeout: 30000,
  pathPrefix: '/v1',
};

export class Api {
  apisauce: ApisauceInstance;

  constructor(config = DEFAULT_CONFIG) {
    this.apisauce = create({
      baseURL: config.url + config.pathPrefix,
      timeout: config.timeout,
      headers: {Accept: 'application/json'},
    });
  }

  async login(phone: string, code: string) {
    const response: ApiResponse<any> = await this.apisauce.post('/auth/login', {
      phone,
      code,
    });
    if (!response.ok) {
      return {kind: 'error' as const, message: response.problem};
    }
    return {kind: 'ok' as const, data: response.data};
  }
}

export const api = new Api();
```

### 七、核心设计模式总结

| 模式                   | 应用场景     | 实现方式                       |
| ---------------------- | ------------ | ------------------------------ |
| **Provider Pattern**   | 全局状态共享 | `RootStoreProvider`            |
| **Singleton Pattern**  | API 实例     | `export const api = new Api()` |
| **HOC Pattern**        | 组件增强     | `observer` 装饰器              |
| **Repository Pattern** | 数据访问     | MST stores                     |

这个项目的架构体现了 **分层架构** 和 **关注点分离** 的原则：

- **UI 层**：screens + components
- **业务层**：hooks + services
- **数据层**：models (MST)
- **配置层**：config + theme
